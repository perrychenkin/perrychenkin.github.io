# calculate weighted score

def calc_weighted_score(y_true, y_pred):
    return (.6 * mean_absolute_error(y_true,y_pred)) - (.3 * (r2_score(y_true, y_pred)*10)) + (.1 * np.sqrt(mse(y_true,y_pred)))

score_results = []
std_results = []
ci_range_results = []
bootstrap_results = {}


years = [2023, 2024, 2025]

df_rfr = df_ml.dropna()

train_df = df_rfr[df_rfr['year'] < 2023]


for combo in metrics_combos:

    cols = list(combo)


    x_train = train_df[cols].to_numpy()
    y_train=train_df['wRC+'].to_numpy()
    
    best_model.fit(x_train, y_train)

    year_results = {}

    for year in years:

        x_test = df_rfr.loc[df_rfr['year'] == year, cols].to_numpy()
        y_test = df_rfr.loc[df_rfr['year'] == year, 'wRC+'].to_numpy()

        bootstrap_scores = np.empty(30)
        y_true_list = []
        y_pred_list = []

        for i in range(30):

            rng = np.random.default_rng(i)
            idx = rng.integers(0,len(x_test),size=len(x_test))
            

            y_pred = best_model.predict(x_test[idx])

            bootstrap_scores[i] = calc_weighted_score(y_test[idx], y_pred)

            y_true_list.append(y_test[idx])
            y_pred_list.append(y_pred)

        bootstrap_scores = np.array(bootstrap_scores)

        year_results[year] = {
            "mean_score": bootstrap_scores.mean(),
            "std": bootstrap_scores.std(),
            "ci_lower": np.percentile(bootstrap_scores, 2.5),
            "ci_upper": np.percentile(bootstrap_scores, 97.5),
            "ci_range": np.percentile(bootstrap_scores, 97.5)
                        - np.percentile(bootstrap_scores, 2.5),
            "scores": bootstrap_scores,
            'y_true':y_true_list,
            'y_pred':y_pred_list  
        }

    bootstrap_results[tuple(cols)] = year_results

    all_scores = np.concatenate([year_results[y]['scores'] for y in years])

    score_results.append({
        "combo": tuple(cols),
        "2023_score": year_results[2023]["mean_score"],
        "2024_score": year_results[2024]["mean_score"],
        "2025_score": year_results[2025]["mean_score"],
        "mean_score": np.mean([year_results[y]["mean_score"] for y in years])
    })

    std_results.append({
        "combo": tuple(cols),
        "2023_std": year_results[2023]["std"],
        "2024_std": year_results[2024]["std"],
        "2025_std": year_results[2025]["std"],
        "total_std": np.std(all_scores)
    })

    ci_range_results.append({
        "combo": tuple(cols),
        "2023_ci_range": year_results[2023]["ci_range"],
        "2024_ci_range": year_results[2024]["ci_range"],
        "2025_ci_range": year_results[2025]["ci_range"],
        "total_ci_range": np.percentile(all_scores, 97.5)
                          - np.percentile(all_scores, 2.5)
    })

# paired bootstrap test

def paired_bootstrap_by_year (combo_a, combo_b, year):
    tuple_a = tuple(combo_a)
    tuple_b = tuple(combo_b)
    
    scores_a = np.array(bootstrap_results[tuple_a][year]['scores'])
    scores_b = np.array(bootstrap_results[tuple_b][year]['scores'])
    
    diff = scores_b - scores_a
    mean_diff = diff.mean()
    
    ci_lower, ci_upper = np.percentile(diff,[2.5,97.5])
    
    return mean_diff, ci_upper, ci_lower

        'combo':top_score
